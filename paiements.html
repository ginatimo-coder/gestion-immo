function chargerPaiements() { fetch('/api/paiements') .then(res => res.json()) .then(data => { const tbody = document.getElementById('paiements-list'); if (data.length === 0) { tbody.innerHTML = 'Aucun paiement enregistré.'; return; } tbody.innerHTML = data.map(p => ` ${p.locataire_nom || ''} ${p.locataire_prenom || ''} ${p.nom_bien || ''} ${p.date_paiement?.substring(0, 10) || ''} ${p.montant || 0} $ ${p.statut || 'Payé'} `).join(''); }) .catch(err => { // C'est ici que la gestion de session expirée doit se trouver, rattachée au fetch console.warn("Session expirée ou non connecté, redirection vers le login..."); window.location.href = '/login'; }); } chargerPaiements();
